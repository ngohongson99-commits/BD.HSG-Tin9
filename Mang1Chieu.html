1. Mảng một chiều là gì?

Mảng một chiều là một dãy các phần tử cùng kiểu dữ liệu, nằm liên tiếp nhau trong bộ nhớ, và được truy cập bằng chỉ số (index). Chỉ số trong C++ bắt đầu từ 0.

5a[0]
8a[1]
2a[2]
9a[3]
4a[4]

Mảng có 5 phần tử thì chỉ số hợp lệ là từ 0 đến 4 (tức n-1).

💡 Ví dụ thực tế: lưu điểm của 30 học sinh, nhiệt độ 7 ngày trong tuần, … thay vì tạo 30 biến riêng lẻ.

2. Mảng tĩnh int a[]

Kích thước mảng phải là hằng số, cố định ngay khi viết chương trình.

Khai báo

int a[5];                    // 5 phần tử, giá trị chưa xác định (rác)
int b[5] = {1, 2, 3, 4, 5};   // khai báo và gán giá trị
int c[5] = {1, 2};            // c = {1, 2, 0, 0, 0} (phần còn lại là 0)
int d[5] = {};                // tất cả bằng 0
int e[] = {7, 8, 9};          // tự suy ra kích thước = 3
const int N = 100;
int f[N];                    // dùng hằng số làm kích thước

Truy cập phần tử

b[0] = 10;            // gán phần tử đầu
int x = b[2];         // đọc phần tử thứ 3

Nhập, xuất mảng

#include <iostream>
using namespace std;

int main() {
    int a[100], n;
    cin >> n;                         // n <= 100
    for (int i = 0; i < n; i++)
        cin >> a[i];                  // nhập
    for (int i = 0; i < n; i++)
        cout << a[i] << " ";          // xuất
    return 0;
}

Duyệt bằng range-for (C++11)

int b[5] = {1, 2, 3, 4, 5};
for (int x : b) cout << x << " ";
⚠️ Mảng tĩnh không tự biết độ dài và không tự kiểm tra chỉ số. Ta phải tự quản lý biến n. Số phần tử khai báo (ví dụ 100) là sức chứa tối đa, còn n là số phần tử đang dùng.

3. Vector vector<int>

Vector là "mảng động" của thư viện chuẩn: tự thay đổi kích thước, biết độ dài của chính nó. Cần #include <vector>.

Khai báo

#include <vector>
using namespace std;

vector<int> v1;                  // rỗng, 0 phần tử
vector<int> v2(5);               // 5 phần tử, đều bằng 0
vector<int> v3(5, 7);            // 5 phần tử, đều bằng 7
vector<int> v4 = {1, 2, 3};      // khởi tạo bằng danh sách
vector<int> v5(v4);              // sao chép từ v4

Các thao tác thường dùng

Thao tácCú phápÝ nghĩa

Kiểm tra rỗngv.empty()true nếu không có phần tử
Truy cậpv[i], v.at(i)at báo lỗi nếu sai chỉ số

Xóa cuốiv.pop_back()Bỏ phần tử cuối

Ví dụ

vector<int> v;
v.push_back(5);
v.push_back(8);
v.push_back(2);                     // v = {5, 8, 2}
cout << v.size();                   // 3
v.insert(v.begin() + 1, 100);       // v = {5, 100, 8, 2}
v.erase(v.begin());                 // v = {100, 8, 2}
v.pop_back();                       // v = {100, 8}

for (int i = 0; i < (int)v.size(); i++) cout << v[i] << " ";
for (int x : v) cout << x << " ";    // cách ngắn gọn hơn

Nhập n số vào vector

int n; cin >> n;
vector<int> v(n);
for (int i = 0; i < n; i++) cin >> v[i];
💡 Sau vector<int> v(n) đã có sẵn n phần tử nên dùng v[i]. Nếu khai báo vector<int> v; (rỗng) thì phải dùng push_back, không được gán v[0] = ....

5. Một số bài toán cơ bản

Mỗi bài có phần mảng tĩnh; phần vector rất tương tự (đổi n thành v.size()).

Bài 1. Tính tổng các phần tử

long long s = 0;
for (int i = 0; i < n; i++) s += a[i];
cout << s;

Bài 2. Tìm giá trị lớn nhất, nhỏ nhất

int mx = a[0], mn = a[0];
for (int i = 1; i < n; i++) {
    if (a[i] > mx) mx = a[i];
    if (a[i] < mn) mn = a[i];
}
cout << mx << " " << mn;

Ý tưởng: giả sử phần tử đầu là lớn nhất, rồi so sánh dần với các phần tử còn lại.

Bài 3. Đếm số chẵn và số lẻ

int chan = 0, le = 0;
for (int i = 0; i < n; i++)
    if (a[i] % 2 == 0) chan++; else le++;

Bài 4. Tìm kiếm tuyến tính

Cho số x, tìm vị trí xuất hiện đầu tiên trong mảng (không có thì in -1).

int pos = -1;
for (int i = 0; i < n; i++)
    if (a[i] == x) { pos = i; break; }
cout << pos;

Bài 5. Đảo ngược mảng

for (int i = 0; i < n / 2; i++)
    swap(a[i], a[n - 1 - i]);

Với vector có thể dùng reverse(v.begin(), v.end()); (thêm #include <algorithm>).

Bài 6. Sắp xếp tăng dần (nổi bọt)

for (int i = 0; i < n - 1; i++)
    for (int j = 0; j < n - 1 - i; j++)
        if (a[j] > a[j + 1]) swap(a[j], a[j + 1]);

Cách nhanh, dùng thư viện:

#include <algorithm>
sort(a, a + n);                 // mảng tĩnh
sort(v.begin(), v.end());       // vector

Bài 7. Chương trình hoàn chỉnh với vector

Nhập n số, in ra tổng, số lớn nhất và các số chẵn.

#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n; cin >> n;
    vector<int> v(n);
    for (int &x : v) cin >> x;

    long long s = 0; int mx = v[0];
    vector<int> chan;
    for (int x : v) {
        s += x;
        if (x > mx) mx = x;
        if (x % 2 == 0) chan.push_back(x);
    }
    cout << "Tong: " << s << "\nMax: " << mx << "\nSo chan: ";
    for (int x : chan) cout << x << " ";
}

Bài tập tự luyện

👉 Đề kèm lời giải nằm ở mục 7.

  1. Tính trung bình cộng các phần tử.
  2. Đếm số lần xuất hiện của giá trị x.
  3. Tìm giá trị lớn thứ hai trong mảng.
  4. Kiểm tra mảng có đối xứng (đọc xuôi ngược giống nhau) không.
  5. Xóa phần tử tại vị trí k của mảng.
  6. Tách các số âm và số dương vào hai vector khác nhau.

6. Lỗi thường gặp

Truy cập ngoài phạm vi: a[5] với mảng 5 phần tử là sai (chỉ số hợp lệ 0..4). Chương trình có thể chạy sai hoặc bị crash mà không báo lỗi rõ ràng.
Quên khởi tạo: mảng khai báo trong hàm int a[5]; chứa giá trị rác. Hãy gán {} hoặc dùng vòng lặp để gán 0 trước khi cộng dồn.
Vòng lặp lệch 1: dùng i <= n thay vì i < n.
Vector rỗng mà dùng v[i]: hãy dùng push_back hoặc khai báo kích thước trước.
So sánh kiểu: i < v.size() có thể cảnh báo vì size() là số không dấu. Viết i < (int)v.size().
Mảng lớn trong main: khai báo như int a[10000000]; trong hàm có thể tràn stack. Hãy khai báo toàn cục hoặc dùng vector.

7. Bài tập có lời giải

Hãy tự làm trước, sau đó bấm Xem lời giải để đối chiếu hướng dẫn và code.

Bài 1. Trung bình cộng

Cho mảng gồm n số nguyên. In ra trung bình cộng của các phần tử (làm tròn 2 chữ số thập phân).

Ví dụ: n = 4, mảng 1 2 3 5 → 2.75

Xem lời giải

Hướng dẫn: tính tổng như bài 1 ở mục 5, rồi chia cho n. Chú ý phải ép kiểu sang double trước khi chia, nếu không phép chia hai số nguyên sẽ cắt phần thập phân.

#include <iostream>
#include <iomanip>
#include <vector>
using namespace std;

int main() {
    int n; cin >> n;
    vector<int> v(n);
    long long s = 0;
    for (int i = 0; i < n; i++) { cin >> v[i]; s += v[i]; }
    cout << fixed << setprecision(2) << (double)s / n;
}

Bài 2. Đếm số lần xuất hiện

Cho mảng n số nguyên và một số x. Đếm xem x xuất hiện bao nhiêu lần.

Ví dụ: mảng 3 5 3 7 3, x = 3 → 3

Xem lời giải

Hướng dẫn: dùng biến đếm cnt = 0. Duyệt từng phần tử, nếu bằng x thì tăng cnt. Khác với tìm kiếm, ở đây không được break vì phải duyệt hết mảng.

int cnt = 0;
for (int i = 0; i < n; i++)
    if (a[i] == x) cnt++;
cout << cnt;

Bài 3. Giá trị lớn thứ hai

Tìm số lớn thứ hai (nhỏ hơn hẳn số lớn nhất) trong mảng. Nếu không có, in Khong co.

Ví dụ: mảng 4 9 9 2 7 → 7; mảng 5 5 5 → Khong co

Xem lời giải

Hướng dẫn: giữ hai biến m1 (lớn nhất) và m2 (lớn thứ hai), khởi tạo rất nhỏ. Khi gặp số lớn hơn m1, đẩy m1 xuống thành m2 rồi cập nhật m1. Ngược lại, nếu số đó nhỏ hơn m1 nhưng lớn hơn m2 thì cập nhật m2. Điều kiện "nhỏ hơn hẳn" giúp bỏ qua các số trùng với max.

#include <climits>

int m1 = INT_MIN, m2 = INT_MIN;
for (int i = 0; i < n; i++) {
    if (a[i] > m1) { m2 = m1; m1 = a[i]; }
    else if (a[i] < m1 && a[i] > m2) m2 = a[i];
}
if (m2 == INT_MIN) cout << "Khong co";
else cout << m2;

Bài 4. Mảng đối xứng

Kiểm tra mảng có đối xứng không (đọc từ trái sang phải và từ phải sang trái giống nhau). In YES hoặc NO.

Ví dụ: 1 2 3 2 1 → YES; 1 2 3 4 → NO

Xem lời giải

Hướng dẫn: so sánh cặp phần tử đối xứng nhau: a[i] với a[n-1-i], chỉ cần duyệt đến giữa mảng (i < n/2). Chỉ cần một cặp khác nhau là kết luận NO.

bool ok = true;
for (int i = 0; i < n / 2; i++) {
    if (a[i] != a[n - 1 - i]) { ok = false; break; }
}
cout << (ok ? "YES" : "NO");

Bài 5. Xóa phần tử tại vị trí k

Cho mảng n số và vị trí k (đánh số từ 0). Xóa phần tử tại vị trí k rồi in mảng mới.

Ví dụ: mảng 5 8 2 9 4, k = 2 → 5 8 9 4

Xem lời giải

Hướng dẫn (mảng tĩnh): dồn các phần tử bên phải k sang trái một ô, rồi giảm n đi 1.

for (int i = k; i < n - 1; i++)
    a[i] = a[i + 1];
n--;
for (int i = 0; i < n; i++) cout << a[i] << " ";

Với vector chỉ cần một dòng:

v.erase(v.begin() + k);
for (int x : v) cout << x << " ";

Bài 6. Tách số âm và số dương

Cho mảng n số nguyên. Đưa các số âm vào một vector, các số dương vào một vector khác (bỏ qua số 0), rồi in hai vector.

Ví dụ: mảng 3 -1 0 -5 8 → âm: -1 -5, dương: 3 8

Xem lời giải

Hướng dẫn: vì chưa biết trước mỗi loại có bao nhiêu số, đây là lúc vector phát huy tác dụng: khai báo hai vector rỗng và dùng push_back để thêm phần tử phù hợp khi duyệt.

vector<int> am, duong;
for (int i = 0; i < n; i++) {
    if (a[i] < 0) am.push_back(a[i]);
    else if (a[i] > 0) duong.push_back(a[i]);
}
cout << "Am: ";
for (int x : am) cout << x << " ";
cout << "\nDuong: ";
for (int x : duong) cout << x << " ";
